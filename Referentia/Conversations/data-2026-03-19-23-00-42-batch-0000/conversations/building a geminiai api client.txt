import time
import os
from watchdog.observers import Observer
from watchdog.events import FileSystemEventHandler
class OrbitalWatcher(FileSystemEventHandler):
    def __init__(self, callback):
        self.callback = callback
        self.last_triggered = 0
    def on_modified(self, event):
        # Prevent "double-fire" noise from certain editors
        current_time = time.time()
        if current_time - self.last_triggered < 2:
            return
            
        if not event.is_directory:
            # Skip hidden files and common ignore folders
            if any(part.startswith('.') for part in 
event.src_path.split(os.sep)):
                return
            if "venv" in event.src_path or "__pycache__" in event.src_path:
                return
            self.last_triggered = current_time
            self.callback(event.src_path)
def start_observer(path, callback):
    """
    Initializes and returns the observer object.
    path: Directory to watch (usually '.')
    callback: The function in orbital.py that handles the logic
    """
    event_handler = OrbitalWatcher(callback)
    observer = Observer()
    observer.schedule(event_handler, path, recursive=True)
    observer.start()
    return observer
