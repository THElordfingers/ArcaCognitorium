```python
    return h

        data["status"] = data["task_status"]

from exc

    data = body.get("data", body)
    task_id = data.get("task_id")
    if not task_id:
        raise ValueError(f"No task_id in response: {resp.text}")

    try:
        resp.raise_for_status()
    except requests.HTTPError as exc:
        raise requests.HTTPError(f"{exc} — body: {resp.text}", response=resp) 
from exc
    body = resp.json()
    data = body.get("data", body)
    return _normalise(data)

    raise TimeoutError(f"Task {task_id} did not complete within
```

```python
"""
FPIcones – api/payload.py
"""

STYLES   = ["solid", "outline", "color", "flat", "sticker"]
FORMATS  = ["png", "svg"]

STEPS_MIN, STEPS_MAX, STEPS_DEFAULT         = 10, 50, 10
GUIDANCE_MIN, GUIDANCE_MAX, GUIDANCE_DEFAULT = 0.0, 10.0, 7.0


def build_icon_payload(
    prompt: str,
    style: str            = "solid",
    fmt: str              = "png",
    num_inference_steps: int  = STEPS_DEFAULT,
    guidance_scale: float     = GUIDANCE_DEFAULT,
) -> dict:
    if not prompt.strip():
        raise ValueError("Prompt must not be empty.")
    if style not in STYLES:
        raise ValueError(f"style must be one of {STYLES}")
    if fmt not in FORMATS:
        raise ValueError(f"format must be one of {FORMATS}")

    payload: dict = {
        "prompt":      prompt.strip(),
        "style":       style,
        "webhook_url": "https://example.com/webhook",  # required by API
    }

    if num_inference_steps != STEPS_DEFAULT:
        payload["num_inference_steps"] = int(num_inference_steps)
    if guidance_scale != GUIDANCE_DEFAULT:
        payload["guidance_scale"] = float(guidance_scale)

    return payload
```
